async function fetchJson(url){
    if (responseCache.has(url)) return responseCache.get(url);
    const attempts = [
        async()=>{const r=await fetch(url);if(!r.ok)throw new Error("direct");return r.json();},
        async()=>{const r=await fetch(`https://api.allorigins.win/get?url=${encodeURIComponent(url)}`);if(!r.ok)throw new Error("allorigins");const j=await r.json();return typeof j.contents==="string"?JSON.parse(j.contents):j.contents;},
        async()=>{const r=await fetch(`https://api.codetabs.com/v1/proxy?quest=${encodeURIComponent(url)}`);if(!r.ok)throw new Error("codetabs");return r.json();},
        async()=>{const r=await fetch(`https://corsproxy.io/?${encodeURIComponent(url)}`);if(!r.ok)throw new Error("corsproxy");return r.json();}
    ];
    for (const attempt of attempts){
        try{
            const data = await attempt();
            if (data && (data.chart || data.news || data.quotes)){responseCache.set(url,data);return data;}
        }catch(e){}
    }
    throw new Error("Market-data services did not respond. Try again or use a different network.");
}

function mean(values){return values.length?values.reduce((a,b)=>a+b,0)/values.length:null;}
function smaAt(values,period){if(values.length<period)return null;return mean(values.slice(-period));}
function emaSeries(values,period){
    if(values.length<period)return [];
    const out=new Array(values.length).fill(null);let seed=mean(values.slice(0,period));out[period-1]=seed;const k=2/(period+1);
    for(let i=period;i<values.length;i++){seed=(values[i]-seed)*k+seed;out[i]=seed;}return out;
}
function emaAt(values,period){const s=emaSeries(values,period);return s.length?s[s.length-1]:null;}
function rsiSeries(values,period=14){
    const out=new Array(values.length).fill(null);if(values.length<=period)return out;
    let gain=0,loss=0;for(let i=1;i<=period;i++){const d=values[i]-values[i-1];gain+=Math.max(d,0);loss+=Math.max(-d,0);}
    let avgGain=gain/period,avgLoss=loss/period;out[period]=avgLoss===0?100:100-(100/(1+avgGain/avgLoss));
    for(let i=period+1;i<values.length;i++){const d=values[i]-values[i-1];avgGain=((avgGain*(period-1))+Math.max(d,0))/period;avgLoss=((avgLoss*(period-1))+Math.max(-d,0))/period;out[i]=avgLoss===0?100:100-(100/(1+avgGain/avgLoss));}
    return out;
}
function atrSeries(high,low,close,period=14){
    const tr=new Array(close.length).fill(null);for(let i=1;i<close.length;i++)tr[i]=Math.max(high[i]-low[i],Math.abs(high[i]-close[i-1]),Math.abs(low[i]-close[i-1]));
    const out=new Array(close.length).fill(null);if(close.length<=period)return out;let value=mean(tr.slice(1,period+1));out[period]=value;
    for(let i=period+1;i<close.length;i++){value=((value*(period-1))+tr[i])/period;out[i]=value;}return out;
}
function macdData(values){
    const fast=emaSeries(values,12),slow=emaSeries(values,26),line=new Array(values.length).fill(null);
    for(let i=0;i<values.length;i++)if(fast[i]!=null&&slow[i]!=null)line[i]=fast[i]-slow[i];
    const compact=line.filter(v=>v!=null),signalCompact=emaSeries(compact,9),signal=new Array(values.length).fill(null);let j=0;
    for(let i=0;i<line.length;i++)if(line[i]!=null){signal[i]=signalCompact[j]??null;j++;}
    const i=values.length-1;return {line:line[i]??0,signal:signal[i]??0,hist:(line[i]??0)-(signal[i]??0),prevHist:i>0&&line[i-1]!=null&&signal[i-1]!=null?line[i-1]-signal[i-1]:0};
}
function adxValue(high,low,close,period=14){
    if(close.length<period*2+1)return null;const tr=[],plusDM=[],minusDM=[];
    for(let i=1;i<close.length;i++){tr.push(Math.max(high[i]-low[i],Math.abs(high[i]-close[i-1]),Math.abs(low[i]-close[i-1])));const up=high[i]-high[i-1],down=low[i-1]-low[i];plusDM.push(up>down&&up>0?up:0);minusDM.push(down>up&&down>0?down:0);}
    let trN=tr.slice(0,period).reduce((a,b)=>a+b,0),plusN=plusDM.slice(0,period).reduce((a,b)=>a+b,0),minusN=minusDM.slice(0,period).reduce((a,b)=>a+b,0);const dx=[];
    for(let i=period;i<tr.length;i++){if(i>period){trN=trN-trN/period+tr[i];plusN=plusN-plusN/period+plusDM[i];minusN=minusN-minusN/period+minusDM[i];}const p=trN?100*plusN/trN:0,m=trN?100*minusN/trN:0;dx.push((p+m)?100*Math.abs(p-m)/(p+m):0);}
    if(dx.length<period)return mean(dx);let adx=mean(dx.slice(0,period));for(let i=period;i<dx.length;i++)adx=((adx*(period-1))+dx[i])/period;return adx;
}
function standardDeviation(values){const m=mean(values);return m==null?null:Math.sqrt(mean(values.map(v=>(v-m)**2)));}
function roc(values,period=20){if(values.length<=period)return 0;const base=values[values.length-1-period];return base?((values[values.length-1]/base)-1)*100:0;}
function slopePct(values,period=10){if(values.length<=period)return 0;const a=values[values.length-1-period],b=values[values.length-1];return a?((b/a)-1)*100/period:0;}

function classifyTimeframe(data){
    let score=0;const reasons=[];
    if(data.ema9&&data.ema21){if(data.price>data.ema9&&data.ema9>data.ema21){score+=2;reasons.push("price > EMA9 > EMA21");}else if(data.price<data.ema9&&data.ema9<data.ema21){score-=2;reasons.push("price < EMA9 < EMA21");}}
    if(data.sma50){score+=data.price>data.sma50?1:-1;}
    if(data.sma200){score+=data.price>data.sma200?1.5:-1.5;}
    if(data.ema21Slope>.03)score+=1;else if(data.ema21Slope<-.03)score-=1;
    if(data.rsi>=52&&data.rsi<=72)score+=1.25;else if(data.rsi<=48&&data.rsi>=28)score-=1.25;else if(data.rsi>78)score-=.25;else if(data.rsi<22)score+=.25;
    if(data.macd.hist>0&&data.macd.hist>=data.macd.prevHist)score+=1.25;else if(data.macd.hist<0&&data.macd.hist<=data.macd.prevHist)score-=1.25;
    if(data.volumeRatio>=1.3)score+=Math.sign(score||data.macd.hist)*.75;
    const direction=score>=2?"BULLISH":score<=-2?"BEARISH":"NEUTRAL";
    const confidence=Math.min(99,Math.round(50+Math.abs(score)*6+(data.adx&&data.adx>25?7:0)));
    return {score,direction,confidence,reasons};
}

function uniqueTerms(terms=[]){return [...new Set(terms.map(x=>String(x).toLowerCase()))];}
function headlineRisk(news=[],profileKey="broad"){
    const titles=news.map(n=>String(n?.title||"").toLowerCase());
    const extra=PROFILE_NEWS_TERMS[profileKey]||{};
    const negativeTerms=uniqueTerms([...GENERIC_NEGATIVE_NEWS,...(extra.negative||[])]),positiveTerms=uniqueTerms([...GENERIC_POSITIVE_NEWS,...(extra.positive||[])]);
    const negative=[],positive=[];
    for(const title of titles){
        for(const term of negativeTerms)if(title.includes(term)&&!negative.includes(term))negative.push(term);
        for(const term of positiveTerms)if(title.includes(term)&&!positive.includes(term))positive.push(term);
    }
    const score=Math.max(-18,Math.min(8,positive.length*2-negative.length*3));
    return {score,negative,positive,count:titles.length};
}

            reasons.unshift(`Signal conflict: bearish ${special.underlying} should support ${ticker}, but ${ticker}'s own chart is bearish. No trade is authorized.`);
        }else if(combined>=20){
            direction="LONG";positionAction=`LONG / HOLD ${ticker}: ${special.underlying} bearish confirmation`;
            reasons.unshift(`${special.underlying} is SHORT; inverse mapping supports a LONG/HOLD bias in ${ticker}.`);
        }else{
            guardrailBlocked=true;combined=clamp(combined,-19,19);positionAction="WAIT: UNDERLYING BEARISH BUT ETF EXECUTION CHART IS NOT CONFIRMED";
        }
    }else{
        guardrailBlocked=true;combined=clamp(combined,-19,19);positionAction=`WAIT / REDUCE RISK: ${special.underlying} has no confirmed direction`;
        reasons.unshift(`${special.underlying} is not a confirmed SHORT, so a new long ${ticker} position is blocked.`);
    }
    // Never recommend opening a short in a daily inverse ETF; bullish-underlying cases are shown as EXIT/AVOID instead.
    if(direction==="SHORT"){direction="WAIT";guardrailBlocked=true;positionAction=`EXIT / AVOID ${ticker} — do not open a contradictory short`;combined=Math.min(combined,19);}
    const confidence=underlyingAnalysis?Math.min(99,Math.round((baseAnalysis.confidence+underlyingAnalysis.confidence)/2)):baseAnalysis.confidence;
    return {...baseAnalysis,direction,score:combined,confidence,guardrailBlocked,reasons,instrumentContext:{ticker,label:special.label,underlying:special.underlying,dailyMultiple:special.dailyMultiple,dailyReset:special.dailyReset,underlyingDirection:underDir,underlyingScore:underlyingAnalysis?.score??0,underlyingConfidence:underlyingAnalysis?.confidence??0,inverseUnderlyingScore,baseDirection:baseAnalysis.direction,baseScore:baseAnalysis.score,positionAction,styleBlocked:!special.allowedStyles.includes(style)}};
}

function buildMacroContext(ticker,profile,{sector=null,vix=null,spy=null,drivers={},news=[]}={}){
    const sectorApplies=document.getElementById("macroGuard")?.checked!==false,econApplies=document.getElementById("econGuard")?.checked!==false,applies=sectorApplies||econApplies;
    if(!applies)return {applies:false,adjustment:0,blockLong:false,blockShort:false,riskLevel:"N/A",reasons:[],profileKey:profile.key,profileLabel:profile.label};
    let adjustment=0,riskPoints=0,tailwindPoints=0;const reasons=[];
    if(!sectorApplies){sector=null;vix=null;spy=null;drivers={};news=[];}
    const sectorTrend=sector?classifyTimeframe(sector).direction:"UNAVAILABLE",spyTrend=spy?classifyTimeframe(spy).direction:"UNAVAILABLE";
    const sectorRelative=sector&&spy?sector.roc20-spy.roc20:null;
    if(sectorTrend==="BEARISH"){adjustment-=12;riskPoints+=2;reasons.push(`${profile.etf} ${profile.label} trend is bearish on the daily chart.`);}
    else if(sectorTrend==="BULLISH"){adjustment+=4;tailwindPoints+=1;reasons.push(`${profile.etf} ${profile.label} trend is bullish on the daily chart.`);}
    if(sectorRelative!=null){
        if(sectorRelative<=-3){adjustment-=6;riskPoints+=1;reasons.push(`${profile.etf} is underperforming SPY by ${Math.abs(sectorRelative).toFixed(1)} percentage points over 20 bars.`);}
        else if(sectorRelative>=3){adjustment+=3;tailwindPoints+=1;reasons.push(`${profile.etf} is outperforming SPY by ${sectorRelative.toFixed(1)} percentage points over 20 bars.`);}
    }
    if(spyTrend==="BEARISH"){
        const penalty=profile.defensive?3:profile.cyclical?8:5;adjustment-=penalty;riskPoints+=profile.cyclical?2:1;reasons.push(`SPY daily trend is bearish${profile.cyclical?", a headwind for this cyclical group":""}.`);
    }else if(spyTrend==="BULLISH"&&profile.cyclical){adjustment+=2;tailwindPoints+=1;}
    const vixPrice=vix?.price??null,vixFactor=profile.defensive?.7:profile.cyclical?1.1:1;
    if(vixPrice!=null){
        if(vixPrice>=30){adjustment-=Math.round(11*vixFactor);riskPoints+=2;reasons.push(`VIX is elevated at ${vixPrice.toFixed(1)}.`);}
        else if(vixPrice>=22){adjustment-=Math.round(5*vixFactor);riskPoints+=1;reasons.push(`VIX is above normal at ${vixPrice.toFixed(1)}.`);}
    }
    const driverReadings=[];
    for(const driver of profile.drivers||[]){
        const data=drivers[driver.symbol]||null,rocValue=data?.roc20??null,vol=data?.price?100*data.atr/data.price:null;
        let impact=0;
        if(rocValue!=null){
            const effective=rocValue*driver.weight;
            if(effective<=-12){impact=-12;riskPoints+=2;}
            else if(effective<=-6){impact=-8;riskPoints+=2;}
            else if(effective<=-2){impact=-4;riskPoints+=1;}
            else if(effective>=12){impact=7;tailwindPoints+=2;}
            else if(effective>=6){impact=4;tailwindPoints+=1;}
            else if(effective>=2){impact=2;tailwindPoints+=1;}
            adjustment+=impact;
            if(Math.abs(impact)>=4)reasons.push(`${driver.label} changed ${signed(rocValue,1)}% over 20 bars — ${impact<0?"macro headwind":"macro tailwind"} for ${profile.label}.`);
        }
        if(vol!=null&&vol>=5&&Math.abs(driver.weight)>=.5){adjustment-=3;riskPoints+=1;reasons.push(`${driver.label} volatility is elevated at ${vol.toFixed(1)}% ATR per bar.`);}
        driverReadings.push({...driver,roc:rocValue,vol,impact});
    }
    const newsRisk=headlineRisk(news,profile.key);adjustment+=newsRisk.score;
    if(newsRisk.negative.length){riskPoints+=Math.min(2,Math.ceil(newsRisk.negative.length/2));reasons.push(`Recent ${profile.label.toLowerCase()} headlines flag: ${newsRisk.negative.slice(0,4).join(", ")}.`);}
    if(newsRisk.positive.length){tailwindPoints+=1;reasons.push(`Positive headline offsets: ${newsRisk.positive.slice(0,3).join(", ")}.`);}
    const economicProbability=calculateEconomicProbability(economicState||ECON_FALLBACK),economicImpact=economicSectorImpact(profile.key,economicProbability);
    if(economicImpact.applies){adjustment+=economicImpact.adjustment;reasons.push(...economicImpact.reasons);if(economicImpact.adjustment<=-8)riskPoints+=2;else if(economicImpact.adjustment<=-4)riskPoints+=1;if(economicImpact.adjustment>=5)tailwindPoints+=1;}
    adjustment=Math.max(-55,Math.min(35,adjustment));
    const riskLevel=riskPoints>=6?"HIGH":riskPoints>=3?"ELEVATED":"NORMAL";
    const bias=adjustment<=-18?"HEADWIND":adjustment>=12?"TAILWIND":"MIXED";
    const blockLong=riskLevel==="HIGH"||adjustment<=-24||economicImpact.blockLong;
    const blockShort=(adjustment>=18&&tailwindPoints>=3&&sectorTrend==="BULLISH")||economicImpact.blockShort;
    if(blockLong)reasons.push(`Sector guardrail blocks a new LONG until the ${profile.label.toLowerCase()} backdrop improves.`);
    if(blockShort)reasons.push(`Sector guardrail blocks a new SHORT because sector and macro conditions are strongly supportive.`);
    return {applies:true,adjustment,blockLong,blockShort,riskLevel,bias,reasons,profileKey:profile.key,profileLabel:profile.label,profileSource:profile.source,sectorETF:profile.etf,sectorTrend,sectorRelative,sectorRoc:sector?.roc20??null,spyTrend,vixPrice,driverReadings,newsRisk,economicProbability,economicImpact};
}

function aggregateAnalysis(frames,benchmarkData,macroContext=null,primaryTimeframe=null){
    const analyses=frames.map((data,i)=>({data,analysis:classifyTimeframe(data),weight:1+i*.35}));
    const weighted=analyses.reduce((sum,x)=>sum+x.analysis.score*x.weight,0),max=analyses.reduce((sum,x)=>sum+8*x.weight,0);
    let normalized=max?weighted/max*100:0;
    const technicalScore=Math.max(-100,Math.min(100,normalized));
    const bullish=analyses.filter(x=>x.analysis.direction==="BULLISH").length,bearish=analyses.filter(x=>x.analysis.direction==="BEARISH").length;
    const alignment=Math.max(bullish,bearish)/analyses.length;
    const primary=frames.find(f=>f.timeframe===(primaryTimeframe||document.getElementById("timeframe").value))||frames.at(-1);
    const relativeStrength=benchmarkData?primary.roc20-benchmarkData.roc20:0;
    if(relativeStrength>2)normalized+=5;else if(relativeStrength<-2)normalized-=5;
    const benchmarkRegime=benchmarkData?classifyTimeframe(benchmarkData):null;
    if(benchmarkRegime?.direction==="BULLISH"&&normalized>0)normalized+=4;
    if(benchmarkRegime?.direction==="BEARISH"&&normalized<0)normalized-=4;
    if(macroContext?.applies)normalized+=macroContext.adjustment;
    normalized=Math.max(-100,Math.min(100,normalized));
    let direction="WAIT";
    if(normalized>=20&&bullish>=2)direction="LONG";
    else if(normalized<=-20&&bearish>=2)direction="SHORT";
    let guardrailBlocked=false;
    if(direction==="LONG"&&macroContext?.blockLong){direction="WAIT";guardrailBlocked=true;normalized=Math.min(normalized,19);}
    if(direction==="SHORT"&&macroContext?.blockShort){direction="WAIT";guardrailBlocked=true;normalized=Math.max(normalized,-19);}
    if(direction==="LONG"&&macroContext?.riskLevel==="ELEVATED"&&(normalized<35||bullish<3)){direction="WAIT";guardrailBlocked=true;normalized=Math.min(normalized,19);}
    let confidence=Math.min(99,Math.round(Math.abs(normalized)*.65+alignment*25+(benchmarkRegime&&benchmarkRegime.direction!=="NEUTRAL"?5:0)));
    if(direction==="WAIT"&&(macroContext?.blockLong||macroContext?.blockShort))confidence=Math.min(confidence,64);
    const reasons=[];
    reasons.push(`${bullish}/${analyses.length} timeframes bullish; ${bearish}/${analyses.length} bearish`);
    reasons.push(`Higher-timeframe alignment: ${Math.round(alignment*100)}%`);
    reasons.push(`Technical-only score before macro filters: ${Math.round(technicalScore)}`);
    if(benchmarkData)reasons.push(`Relative strength vs ${benchmarkData.ticker}: ${signed(relativeStrength,2)}% over 20 bars`);
    else reasons.push("Benchmark data unavailable; market-regime and relative-strength filters were not applied.");
    if(benchmarkRegime)reasons.push(`${benchmarkData.ticker} regime: ${benchmarkRegime.direction} (${TF_LABELS[benchmarkData.timeframe]})`);
    if(macroContext?.applies){reasons.push(`${macroContext.profileLabel} macro adjustment: ${macroContext.adjustment>0?"+":""}${macroContext.adjustment} points (${macroContext.riskLevel} risk, ${macroContext.bias.toLowerCase()} bias)`);reasons.push(`Sector classification: ${macroContext.profileLabel} via ${macroContext.profileSource}.`);reasons.push(...macroContext.reasons);}
    if(primary.adx!=null)reasons.push(`Trend strength ADX: ${primary.adx.toFixed(1)} (${primary.adx>=25?"trending":"weak/range-bound"})`);
    reasons.push(`Primary volume: ${primary.volumeRatio.toFixed(2)}× 20-bar average`);
    return {direction,score:Math.round(normalized),technicalScore:Math.round(technicalScore),confidence,analyses,alignment,relativeStrength,benchmarkRegime,primary,reasons,macroContext,guardrailBlocked};
}

function marketRegime(data){
    if(!data)return "UNAVAILABLE";
    if(data.price>(data.sma50||data.ema21)&&(!data.sma200||data.sma50>data.sma200))return "RISK-ON";
    if(data.price<(data.sma50||data.ema21)&&(!data.sma200||data.sma50<data.sma200))return "RISK-OFF";
    return "MIXED";
}

function calculateTrade(data,analysis){
    const account=Math.max(0,Number(document.getElementById("account").value)||0),riskPct=clamp(Number(document.getElementById("risk").value)||1,.1,5),maxPositionPct=clamp(Number(document.getElementById("maxPosition").value)||20,1,100);
    const atrMult=clamp(Number(document.getElementById("atrMultiple").value)||1.7,.5,5),maxStopPct=clamp(Number(document.getElementById("maxStopPct").value)||8,1,25),rewardMult=clamp(Number(document.getElementById("rewardMultiple").value)||2.5,1,10),slippagePct=clamp(Number(document.getElementById("slippage").value)||0,0,3)/100;
    const isShort=analysis.direction==="SHORT",isTrade=analysis.direction!=="WAIT",marketEntry=isShort?data.price*(1-slippagePct):data.price*(1+slippagePct);
    const anchor=isShort?Math.min(marketEntry,data.ema9||marketEntry):Math.max(marketEntry,data.ema9||marketEntry);

    // Use ATR plus recent structure, but never allow an old weekly/monthly high or bad data point
    // to create an unusable stop. Example: a $14 short with an 8% cap cannot exceed $15.12.
    const atrDistance=Math.max(data.atr*atrMult,anchor*.003);
    const rawStructuralStop=isShort?Math.max(data.resistance+data.atr*.10,anchor+atrDistance):Math.min(data.support-data.atr*.10,anchor-atrDistance);
    const rawStopDistance=Math.abs(anchor-rawStructuralStop);
    const maxStopDistance=anchor*(maxStopPct/100);
    const stopWasCapped=Number.isFinite(rawStopDistance)&&rawStopDistance>maxStopDistance;
    let stopDistance=Number.isFinite(rawStopDistance)?Math.min(rawStopDistance,maxStopDistance):Math.min(atrDistance,maxStopDistance);
    stopDistance=Math.max(stopDistance,anchor*.003);

    let stop=isShort?anchor+stopDistance:anchor-stopDistance;
    if(!isShort && stop>=anchor)stop=Math.max(.01,anchor-stopDistance);
    if(isShort && stop<=anchor)stop=anchor+stopDistance;
    stopDistance=Math.abs(anchor-stop);

    const t1=Math.max(.01,isShort?anchor-stopDistance:anchor+stopDistance);
    const t2=Math.max(.01,isShort?anchor-stopDistance*rewardMult:anchor+stopDistance*rewardMult);
    const runnerR=Math.max(3,rewardMult+1);
    const t3=Math.max(.01,isShort?anchor-stopDistance*runnerR:anchor+stopDistance*runnerR);
    const dollarRisk=account*riskPct/100,riskShares=stopDistance?Math.floor(dollarRisk/stopDistance):0,allocationShares=anchor?Math.floor((account*maxPositionPct/100)/anchor):0;
    const liquidityCap=data.avgDollarVolume?Math.floor((data.avgDollarVolume*.01)/anchor):Infinity;
    const shares=isTrade?Math.max(0,Math.min(riskShares,allocationShares,liquidityCap)):0;
    const stopPct=anchor?stopDistance/anchor*100:0;
    const t1Pct=anchor?Math.abs(t1-anchor)/anchor*100:0;
    const t2Pct=anchor?Math.abs(t2-anchor)/anchor*100:0;
    const t3Pct=anchor?Math.abs(t3-anchor)/anchor*100:0;
    const rr1=stopDistance?Math.abs(t1-anchor)/stopDistance:0;
    const rr2=stopDistance?Math.abs(t2-anchor)/stopDistance:0;
    const rr3=stopDistance?Math.abs(t3-anchor)/stopDistance:0;
    const warnings=[];
    if(stopWasCapped)warnings.push(`Raw structural stop ${money(rawStructuralStop)} was rejected as too far away. Stop distance was capped at ${maxStopPct.toFixed(1)}% of entry.`);
    if(data.avgDollarVolume<10_000_000)warnings.push("Liquidity warning: average dollar volume is below $10 million.");
    if(data.volumeRatio<.7)warnings.push("Current volume is below 70% of its 20-bar average.");
    if(stopPct>=maxStopPct-.01)warnings.push(`Stop is at the selected ${maxStopPct.toFixed(1)}% maximum; consider waiting for a tighter entry or reducing size.`);
    if(data.adx!=null&&data.adx<18)warnings.push("ADX indicates a weak or range-bound trend; breakout signals have lower reliability.");
    if(!isShort&&data.price>data.bbUpper)warnings.push("Long entry is extended above the upper volatility band; avoid chasing and wait for a controlled entry.");
    if(isShort&&data.price<data.bbLower)warnings.push("Short entry is extended below the lower volatility band; avoid chasing and wait for a controlled entry.");
    if(Math.abs(data.gapPct)>=4)warnings.push(`Large opening gap detected (${signed(data.gapPct,2)}%); spreads and reversal risk may be elevated.`);
    if(analysis.macroContext?.applies&&(analysis.macroContext.riskLevel!=="NORMAL"||Math.abs(analysis.macroContext.adjustment)>=8))warnings.push(`${analysis.macroContext.profileLabel} sector guardrail: ${analysis.macroContext.riskLevel} risk with a ${analysis.macroContext.adjustment} point score adjustment.`);
    if(analysis.direction==="WAIT")warnings.push(analysis.macroContext?.blockLong?`No LONG is authorized because ${analysis.macroContext.profileLabel} macro conditions conflict with the technical setup.`:analysis.macroContext?.blockShort?`No SHORT is authorized because ${analysis.macroContext.profileLabel} macro conditions strongly support the stock.`:"No trade is authorized because multi-timeframe confirmation is insufficient.");
    if(shares===0&&isTrade)warnings.push("Position size calculated as zero under the current risk and allocation limits.");
    return {entry:anchor,entryLow:isShort?anchor:Math.max(0,anchor-data.atr*.15),entryHigh:isShort?anchor+data.atr*.15:anchor,stop,t1,t2,t3,stopDistance,stopPct,t1Pct,t2Pct,t3Pct,rr1,rr2,rr3,shares,positionValue:shares*anchor,estimatedRisk:shares*stopDistance,riskBudget:dollarRisk,rewardMult,runnerR,maxStopPct,rawStructuralStop,stopWasCapped,warnings,isShort,isTrade};
}

async function fetchTickerSnapshot(query){

        const raw=await fetchJson(`https://query1.finance.yahoo.com/v1/finance/search?q=${encodeURIComponent(query)}&quotesCount=5&newsCount=8`);

}

function evaluatePrimeCandidate(analysis,trade,profile,thresholds){
    const d=analysis.primary,m=analysis.macroContext||{},dir=analysis.direction,failures=[],warnings=[];
    const alignmentPct=analysis.alignment*100;
    const stockVsSector=m.sectorRoc==null?null:d.roc20-m.sectorRoc;
    const supportValid=Number.isFinite(d.support)&&d.support>0&&d.support<=d.price&&d.price/d.support<3;
    const resistanceValid=Number.isFinite(d.resistance)&&d.resistance>=d.price&&d.resistance/d.price<3;
    const stopOrientation=dir==="LONG"?trade.stop<trade.entry:dir==="SHORT"?trade.stop>trade.entry:false;
    if(getSpecialInstrument(d.ticker)?.primeEligible===false)failures.push("Daily-reset leveraged/inverse ETF excluded from Prime scanner; analyze its underlying instead");
    if(!["LONG","SHORT"].includes(dir))failures.push("No actionable LONG/SHORT signal");
    if(Math.abs(analysis.score)<thresholds.minScore)failures.push(`Score below ${thresholds.minScore}`);
    if(analysis.confidence<thresholds.minConfidence)failures.push(`Confidence below ${thresholds.minConfidence}%`);
    if(alignmentPct<thresholds.minAlignment)failures.push(`Alignment below ${thresholds.minAlignment}%`);
    if(d.volumeRatio<thresholds.minVolume)failures.push(`Volume below ${thresholds.minVolume.toFixed(2)}×`);
    if((d.adx??0)<thresholds.minAdx)failures.push(`ADX below ${thresholds.minAdx}`);
    if(d.avgDollarVolume<thresholds.minDollarVolume)failures.push(`Dollar volume below ${compactMoney(thresholds.minDollarVolume)}`);
    if(trade.rr2<thresholds.minRR)failures.push(`Reward/risk below ${thresholds.minRR.toFixed(2)}:1`);
    if(trade.stopWasCapped)failures.push("Structural stop required capping");
    if(!stopOrientation)failures.push("Stop is on the wrong side of entry");
    if(!supportValid||!resistanceValid)failures.push("Support/resistance data failed sanity check");
    if(dataAgeHours(d)>timeframeMaxAgeHours(d.timeframe))failures.push("Price data may be stale");
    if(!Number.isFinite(d.price)||d.price<3)failures.push("Price below $3 or invalid");
    if(trade.shares<1)failures.push("Position sizing produced zero shares");
    if(profile.source==="broad-market fallback")failures.push("Sector classification could not be verified");
    if(document.getElementById("macroGuard")?.checked!==false&&m.sectorTrend==="UNAVAILABLE")failures.push("Sector ETF data unavailable");
    if(document.getElementById("macroGuard")?.checked!==false&&m.vixPrice==null)warnings.push("VIX data unavailable");
    if(dir==="LONG"){
        if(d.rsi>76)failures.push("Long is overbought above RSI 76");
        if(d.price>d.bbUpper*1.025)failures.push("Long is extended above its volatility band");
        if(analysis.relativeStrength<0)failures.push("Stock is underperforming benchmark");
        if(stockVsSector!=null&&stockVsSector<0)failures.push("Stock is underperforming its sector ETF");
        if(m.riskLevel==="HIGH"||m.blockLong||m.adjustment<=-8)failures.push("Macro/sector backdrop conflicts with LONG");
        if(m.sectorTrend==="BEARISH")failures.push("Sector ETF trend is bearish");
    }
    if(dir==="SHORT"){
        if(d.rsi<24)failures.push("Short is oversold below RSI 24");
        if(d.price<d.bbLower*.975)failures.push("Short is extended below its volatility band");
        if(analysis.relativeStrength>0)failures.push("Stock is outperforming benchmark");
        if(stockVsSector!=null&&stockVsSector>0)failures.push("Stock is outperforming its sector ETF");
        if(m.blockShort||m.adjustment>=8)failures.push("Macro/sector backdrop conflicts with SHORT");
        if(m.sectorTrend==="BULLISH"&&analysis.score>-75)failures.push("Sector ETF trend is bullish");
    }
    const uniqueFailures=[...new Set(failures)],uniqueWarnings=[...new Set(warnings)];
    const volumeComponent=Math.min(10,(d.volumeRatio/Math.max(thresholds.minVolume,.1))*7);
    const adxComponent=Math.min(10,((d.adx||0)/Math.max(thresholds.minAdx,1))*7);
    const relativeComponent=Math.min(8,Math.abs(analysis.relativeStrength)*1.5);
    const quality=Math.round(clamp(Math.abs(analysis.score)*.36+analysis.confidence*.24+alignmentPct*.18+volumeComponent+adxComponent+relativeComponent,0,100));
    const passed=uniqueFailures.length===0;
    const nearPrime=!passed&&uniqueFailures.length<=2&&Math.abs(analysis.score)>=thresholds.minScore-12&&analysis.confidence>=thresholds.minConfidence-10&&["LONG","SHORT"].includes(dir);
    return {passed,nearPrime,quality,failures:uniqueFailures,warnings:uniqueWarnings,alignmentPct,stockVsSector};
}

        if(runId!==primeScanRunId)return;
        try{
            if(getSpecialInstrument(ticker)?.primeEligible===false&&document.getElementById("leverageGuard")?.checked!==false){errors.push({ticker,error:"Excluded daily-reset leveraged/inverse ETF; scan underlying instead"});completed++;updatePrimeProgress(completed/universe.length*38,`Stage 1 of 2: ${completed}/${universe.length} symbols screened.`);return;}
            const data=await loadTickerData(ticker,primary),signal=classifyTimeframe(data),direction=signal.direction==="BULLISH"?"LONG":signal.direction==="BEARISH"?"SHORT":"WAIT";
            const liquid=data.avgDollarVolume>=thresholds.minDollarVolume*.5,strongEnough=Math.abs(signal.score)>=1.75&&(data.adx??0)>=Math.max(14,thresholds.minAdx-8);
            if(direction!=="WAIT"&&candidateDirectionAllowed(direction)&&liquid&&strongEnough){
                const rank=Math.abs(signal.score)*12+(data.adx||0)+Math.min(data.volumeRatio,2)*8+Math.abs(data.roc20);
                quick.push({ticker,direction,rank,data});
            }
        }catch(e){errors.push({ticker,error:e.message});}
        completed++;updatePrimeProgress(completed/universe.length*38,`Stage 1 of 2: ${completed}/${universe.length} symbols screened.`);
    });
    if(runId!==primeScanRunId)return;
    const maxDeepPerSide=Math.min(18,Math.max(8,thresholds.maxResults*2));
    const longPool=quick.filter(x=>x.direction==="LONG").sort((a,b)=>b.rank-a.rank).slice(0,maxDeepPerSide);
    const shortPool=quick.filter(x=>x.direction==="SHORT").sort((a,b)=>b.rank-a.rank).slice(0,maxDeepPerSide);
    let deepPool=document.getElementById("primeDirection").value==="long"?longPool:document.getElementById("primeDirection").value==="short"?shortPool:[...longPool,...shortPool];
    deepPool=[...new Map(deepPool.map(x=>[x.ticker,x])).values()];
    if(!deepPool.length){
        updatePrimeProgress(100,"No symbols passed the fast technical/liquidity prefilter.");
        document.getElementById("primeSummary").innerHTML=[tile("Universe",String(universe.length),"blue"),tile("Deep Candidates","0","yellow"),tile("Data Errors",String(errors.length),errors.length?"red":"green")].join("");
        renderPrimePassed([],"primeLongOutput");renderPrimePassed([],"primeShortOutput");renderPrimeWatch([]);button.disabled=false;setStatus("Prime scan complete");return;
    }
    updatePrimeProgress(40,`Stage 2 of 2: loading full multi-timeframe, sector, macro, and headline data for ${deepPool.length} finalists.`);
    const profileSeed=Object.fromEntries(deepPool.map(x=>[x.ticker,resolveSectorProfile(x.ticker,null,{ignoreOverride:true})]));
    const neededSymbols=[...new Set(deepPool.flatMap(x=>[profileSeed[x.ticker].etf,...profileSeed[x.ticker].drivers.map(d=>d.symbol)]).concat(["^VIX","SPY"]))];
    const macroPairs=document.getElementById("macroGuard").checked?await mapLimit(neededSymbols,3,async symbol=>[symbol,await loadTickerData(symbol,"1d").catch(()=>null)]):[];
    const macroMap=Object.fromEntries(macroPairs),vix=macroMap["^VIX"]||null,spy=macroMap.SPY||null;
    const benchmarkSymbol=document.getElementById("benchmark").value;
    const benchmarkResult=await loadTickerData(benchmarkSymbol,primary).catch(()=>null);
    const results=[];completed=0;
    await mapLimit(deepPool,2,async seed=>{
        if(runId!==primeScanRunId)return;
        try{
            const [frameResults,snapshot]=await Promise.all([
                mapLimit(frames,2,async tf=>{try{return await loadTickerData(seed.ticker,tf)}catch(e){return {error:e.message,timeframe:tf}}}),
                fetchTickerSnapshot(seed.ticker)
            ]);
            const cleanFrames=frameResults.filter(x=>!x.error);
            if(cleanFrames.length<Math.min(3,frames.length))throw new Error("Insufficient multi-timeframe data");
            const profile=resolveSectorProfile(seed.ticker,snapshot.quote,{ignoreOverride:true}),drivers={};
            for(const d of profile.drivers)drivers[d.symbol]=macroMap[d.symbol]||null;
            const macro=buildMacroContext(seed.ticker,profile,{sector:macroMap[profile.etf]||null,vix,spy,drivers,news:snapshot.news});
            const analysis=aggregateAnalysis(cleanFrames,benchmarkResult,macro,primary),trade=calculateTrade(analysis.primary,analysis),evaluation=evaluatePrimeCandidate(analysis,trade,profile,thresholds);
            results.push({ticker:seed.ticker,profile,macro,analysis,trade,evaluation});
        }catch(e){errors.push({ticker:seed.ticker,error:e.message});}
        completed++;updatePrimeProgress(40+completed/deepPool.length*58,`Stage 2 of 2: ${completed}/${deepPool.length} finalists fully analyzed.`);
    });
    if(runId!==primeScanRunId)return;
    const longs=results.filter(r=>r.evaluation.passed&&r.analysis.direction==="LONG").sort((a,b)=>b.evaluation.quality-a.evaluation.quality).slice(0,thresholds.maxResults);
    const shorts=results.filter(r=>r.evaluation.passed&&r.analysis.direction==="SHORT").sort((a,b)=>b.evaluation.quality-a.evaluation.quality).slice(0,thresholds.maxResults);
    const watch=results.filter(r=>r.evaluation.nearPrime).sort((a,b)=>b.evaluation.quality-a.evaluation.quality).slice(0,Math.max(10,thresholds.maxResults*2));
    renderPrimePassed(longs,"primeLongOutput");renderPrimePassed(shorts,"primeShortOutput");renderPrimeWatch(watch);
    document.getElementById("primeSummary").innerHTML=[
        tile("Universe Scanned",String(universe.length),"blue"),
        tile("Deep Finalists",String(deepPool.length),"purple"),
        tile("Prime Longs",String(longs.length),longs.length?"green":"yellow"),
        tile("Prime Shorts",String(shorts.length),shorts.length?"red":"yellow"),
        tile("Near Prime",String(watch.length),watch.length?"yellow":"blue"),
        tile("Data Errors",String(errors.length),errors.length?"red":"green"),
        tile("Scanner Timeframes",frames.map(tf=>TF_LABELS[tf]).join(" / "),"cyan")
    ].join("");
    updatePrimeProgress(100,`Complete: ${longs.length} prime longs, ${shorts.length} prime shorts, and ${watch.length} near-prime candidates.`);
    setStatus("Prime scan complete");
    }catch(error){
        console.error(error);
        updatePrimeProgress(100,`Prime scan failed: ${error.message}`);
        document.getElementById("primeSummary").innerHTML=[tile("Status","Scan failed","red"),tile("Reason",String(error.message||"Unknown error"),"yellow")].join("");
        setStatus("Prime scan failed","var(--red)");
    }finally{
        button.disabled=false;
    }
}

            analysis=aggregateAnalysis(frames,benchmarkResult,macroContext);
        }
        const trade=calculateTrade(analysis.primary,analysis);
        renderChart(analysis.primary,trade);renderRegime(analysis,benchmarkResult);renderMTF(analysis);renderTradeOutput(analysis,trade);renderPositionGuardrails(ticker,analysis.primary,analysis);renderNews(displayNews,ticker);

Public proxy services can temporarily rate-limit requests.</div>`;document.getElementById("positionCard").style.display="none";setStatus("Analysis failed","var(--red)");}

}

async function runScanner(){
    const tickers=[...new Set(document.getElementById("watchlist").value.split(",").map(x=>x.trim().toUpperCase().replace(/[^A-Z0-9.\-^=]/g,"")).filter(Boolean))].slice(0,12);
    const card=document.getElementById("screenerCard"),out=document.getElementById("screenerOutput"),bar=document.querySelector("#scanProgress>div"),btn=document.getElementById("scanBtn");card.style.display="block";
    if(!tickers.length){out.innerHTML='<div class="red">Enter at least one ticker.</div>';return;}
    btn.disabled=true;out.innerHTML=`<div class="loading">Ranking ${tickers.length} symbols with sector guardrails…</div>`;bar.style.width="0%";setStatus("Watchlist scan running…","var(--yellow)");
    const pack=getTimeframes(),fastTF=document.getElementById("timeframe").value,anchorTF=pack.at(-1),rows=[];let completed=0;
    const profiles=Object.fromEntries(tickers.map(t=>[t,resolveSectorProfile(t,null,{ignoreOverride:true})]));
    const guardOn=document.getElementById("macroGuard").checked;
    const neededSymbols=guardOn?[...new Set(tickers.flatMap(t=>[profiles[t].etf,...profiles[t].drivers.map(d=>d.symbol)]).concat(["^VIX","SPY"]))]:[];
    const macroPairs=guardOn?await mapLimit(neededSymbols,3,async symbol=>[symbol,await loadTickerData(symbol,"1d").catch(()=>null)]):[];
    const macroMap=Object.fromEntries(macroPairs),vix=macroMap["^VIX"]||null,spy=macroMap.SPY||null;
    await mapLimit(tickers,2,async ticker=>{
        try{
            const special=getSpecialInstrument(ticker);
            if(special&&document.getElementById("leverageGuard")?.checked!==false){
                const fast=await loadTickerData(ticker,fastTF);rows.push({ticker,sector:`Daily inverse of ${special.underlying}`,price:fast.price,signal:"UNDERLYING",score:0,macro:0,confidence:0,rsi:fast.rsi,adx:fast.adx,volume:fast.volumeRatio,roc:fast.roc20});completed++;bar.style.width=`${completed/tickers.length*100}%`;return;
            }
            const [fast,anchor]=await Promise.all([loadTickerData(ticker,fastTF),fastTF===anchorTF?Promise.resolve(null):loadTickerData(ticker,anchorTF)]),frames=anchor?[fast,anchor]:[fast],profile=profiles[ticker],drivers={};
            for(const d of profile.drivers)drivers[d.symbol]=macroMap[d.symbol]||null;
            const macro=buildMacroContext(ticker,profile,{sector:macroMap[profile.etf]||null,vix,spy,drivers,news:[]}),pseudo=aggregateAnalysis(frames,null,macro);
            rows.push({ticker,sector:profile.label,price:fast.price,signal:pseudo.direction,score:pseudo.score,macro:macro.adjustment,confidence:pseudo.confidence,rsi:fast.rsi,adx:fast.adx,volume:fast.volumeRatio,roc:fast.roc20});
        }catch(e){rows.push({ticker,error:e.message});}
        completed++;bar.style.width=`${completed/tickers.length*100}%`;
    });
    rows.sort((a,b)=>Math.abs(b.score||0)-Math.abs(a.score||0));
    out.innerHTML=`<table><thead><tr><th>Rank</th><th>Ticker</th><th>Sector</th><th>Signal</th><th>Score</th><th>Macro</th><th>Confidence</th><th>Price</th><th>RSI</th><th>ADX</th><th>Volume</th><th>ROC 20</th></tr></thead><tbody>${rows.map((r,i)=>r.error?`<tr><td>${i+1}</td><td>${r.ticker}</td><td colspan="10" class="red">Data unavailable</td></tr>`:`<tr><td>${i+1}</td><td><button class="btn-sm" onclick="document.getElementById('ticker').value='${r.ticker}';runEngine()">${r.ticker}</button></td><td>${escapeHTML(r.sector)}</td><td class="${r.signal==="LONG"?"green":r.signal==="SHORT"?"red":"yellow"}">${r.signal==="UNDERLYING"?"▶ ANALYZE IREN FIRST":`${directionArrow(r.signal)} ${r.signal}`}</td><td>${r.score>0?"+":""}${r.score}</td><td class="${r.macro>0?"green":r.macro<0?"red":"yellow"}">${r.macro>0?"+":""}${r.macro}</td><td>${r.confidence}%</td><td>${money(r.price)}</td><td>${r.rsi.toFixed(1)}</td><td>${r.adx==null?"—":r.adx.toFixed(1)}</td><td>${r.volume.toFixed(2)}×</td><td class="${r.roc>=0?"green":"red"}">${signedArrow(r.roc,2)}%</td></tr>`).join("")}</tbody></table>`;

function clamp(v,min,max){return Math.max(min,Math.min(max,v));}
function money(v){return Number.isFinite(v)?v.toLocaleString(undefined,{style:"currency",currency:"USD",minimumFractionDigits:2,maximumFractionDigits:2}):"—";}
function compactMoney(v){if(!Number.isFinite(v))return"—";return new Intl.NumberFormat(undefined,{style:"currency",currency:"USD",notation:"compact",maximumFractionDigits:1}).format(v);}
function percent(v,d=2){return Number.isFinite(v)?`${Number(v).toFixed(d)}%`:"—";}
function directionArrow(direction){return ["LONG","BULLISH","RISK-ON"].includes(direction)?"▲":["SHORT","BEARISH","RISK-OFF"].includes(direction)?"▼":"▶";}
function arrowClass(direction){return ["LONG","BULLISH","RISK-ON"].includes(direction)?"dir-up":["SHORT","BEARISH","RISK-OFF"].includes(direction)?"dir-down":"dir-flat";}
function signedArrow(v,d=1){const n=Number(v||0);return `${n>0?"▲":n<0?"▼":"▶"} ${n>=0?"+":""}${n.toFixed(d)}`;}
function signed(v,d=1){return `${v>=0?"+":""}${Number(v||0).toFixed(d)}`;}
function formatDateTime(unix){return Number.isFinite(unix)?new Date(unix*1000).toLocaleString(undefined,{timeZone:"America/Chicago",timeZoneName:"short"}):"—";}
function escapeHTML(value){return String(value??"").replace(/[&<>'"]/g,c=>({"&":"&amp;","<":"&lt;",">":"&gt;","'":"&#39;",'"':"&quot;"}[c]));}
function safeUrl(url){try{const u=new URL(url);return ["http:","https:"].includes(u.protocol)?u.href:"#";}catch(e){return"#";}}
